STEP 1 — Disable the GP Elements hook (so it doesn't duplicate this)

Go to Elements, open "Topic Archive Hero," and set its Status to Draft, then Update. This
stops its output everywhere (it was never firing on the search-within-topic page anyway,
so this changes nothing there — but on the normal topic archive pages it WAS firing, so
without this step you'd get the hero wrapped twice).

STEP 2 — Paste this at the end of functions.php

This is the exact same style/script content from the GP Elements hook, unchanged, just
moved into a PHP function hooked on is_tax('topic') directly instead of relying on GP
Premium's own "Post Topic Archive" condition matching — which is what was silently
skipping this whole block whenever a search term (?s=) was also in the URL, even though
the page is still genuinely a topic archive. is_tax('topic') is confirmed true in both
states, so this fires correctly either way.
*/

<script>
document.addEventListener('DOMContentLoaded', function () {
  var hero = document.querySelector('.site-main > div:first-child');
  if (!hero || hero.querySelector('.spt-hero-search')) return;
  var h1 = hero.querySelector('h1');
  var topicName = h1 ? h1.textContent.trim() : '';

  var inner = document.createElement('div');
  inner.className = 'spt-hero-inner';

  var left = document.createElement('div');
  left.className = 'spt-hero-left';
  while (hero.firstChild) {
    left.appendChild(hero.firstChild);
  }
  var eyebrow = left.children[0];
  if (eyebrow && eyebrow.textContent.indexOf('/') !== -1) {
    var parts = eyebrow.textContent.split('/');
    eyebrow.innerHTML = parts[0].trim() + ' <span class="spt-eyebrow-topic">/ ' + parts[1].trim() + '</span>';
  }
  inner.appendChild(left);

  var right = document.createElement('div');
  right.className = 'spt-hero-search';
  right.innerHTML =
    '<p class="spt-search-label">Search This Topic</p>' +
    '<form class="spt-search-form" action="" method="get">' +
      '<svg class="spt-search-icon" viewBox="0 0 24 24" width="16" height="16"><circle cx="11" cy="11" r="7" fill="none" stroke="currentColor" stroke-width="2"/><line x1="16.5" y1="16.5" x2="21" y2="21" stroke="currentColor" stroke-width="2" stroke-linecap="round"/></svg>' +
      '<input type="search" name="s" placeholder="Search ' + topicName + '" />' +
    '</form>';
  inner.appendChild(right);

  hero.appendChild(inner);

  var navLinks = document.querySelectorAll('#site-navigation a');
  navLinks.forEach(function (a) {
    if (a.textContent.trim().indexOf('Perspective') === 0) {
      a.classList.add('spt-topic-active');
      a.setAttribute('aria-current', 'page');
    }
  });

  if (!document.querySelector('.spt-nav-cta-item')) {
    var navUl = document.querySelector('#site-navigation ul');
    if (navUl) {
      var li = document.createElement('li');
      li.className = 'spt-nav-cta-item';
      var a = document.createElement('a');
      a.href = '/schedule-a-conversation/';
      a.target = '_blank';
      a.rel = 'noopener';
      a.textContent = 'Schedule a Conversation';
      li.appendChild(a);
      navUl.appendChild(li);
    }
  }
});
